cmake_minimum_required(VERSION 3.16)
project(channel_dns LANGUAGES Fortran)
option(ENABLE_UNIT_TESTS "Build tests" ON)
option(BUILD_LEGACY_REFERENCE "Build the FFTPACK reference solver" ON)
find_package(MPI REQUIRED COMPONENTS Fortran)
set(CHANNEL_DNS_FFT_BACKEND "FFTW" CACHE STRING "FFT backend: FFTW or FFTPACK")
set_property(CACHE CHANNEL_DNS_FFT_BACKEND PROPERTY STRINGS FFTW FFTPACK)
if(CHANNEL_DNS_FFT_BACKEND STREQUAL "FFTW")
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(FFTW REQUIRED IMPORTED_TARGET fftw3)
 find_path(FFTW_FORTRAN_INCLUDE_DIR fftw3.f03 HINTS ${FFTW_INCLUDE_DIRS})
 if(NOT FFTW_FORTRAN_INCLUDE_DIR)
  message(FATAL_ERROR "FFTW Fortran interface fftw3.f03 is required")
 endif()
elseif(NOT CHANNEL_DNS_FFT_BACKEND STREQUAL "FFTPACK")
 message(FATAL_ERROR "CHANNEL_DNS_FFT_BACKEND must be FFTW or FFTPACK")
endif()
if(ENABLE_UNIT_TESTS)
  enable_testing()
endif()

add_subdirectory(src)
if(BUILD_LEGACY_REFERENCE)
  add_subdirectory(legacy)
endif()
